cmake_minimum_required(VERSION 3.24)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 61)
endif()
project(guild_kernels_cuda VERSION 1.4.0 LANGUAGES C CXX CUDA)
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
  OUTPUT_VARIABLE PACKAGE_SOURCE_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
find_package(GuildKernelSDK 1.4 REQUIRED CONFIG)
find_package(CUDAToolkit REQUIRED)
include(FetchContent)
FetchContent_Declare(cuda_ggml_source
  GIT_REPOSITORY https://github.com/ggml-org/llama.cpp.git
  GIT_TAG 3cf03257f219afbe7334045ff7c6a06ac68c627d GIT_SHALLOW FALSE)
FetchContent_GetProperties(cuda_ggml_source)
if(NOT cuda_ggml_source_POPULATED)
  FetchContent_Populate(cuda_ggml_source)
endif()
include(cmake/GuildCUDADenseSources.cmake)
add_library(cuda_dense_math STATIC)
guild_cuda_attach_dense(cuda_dense_math)
target_link_libraries(cuda_dense_math PUBLIC Guild::KernelSDK CUDA::cudart)
add_library(cuda_attn_math STATIC)
guild_cuda_attach_attn(cuda_attn_math)
target_link_libraries(cuda_attn_math PUBLIC Guild::KernelSDK CUDA::cudart)
add_library(cuda_moe_math STATIC)
guild_cuda_attach_moe(cuda_moe_math)
target_link_libraries(cuda_moe_math PUBLIC Guild::KernelSDK CUDA::cudart)
add_library(cuda_iq_math STATIC src/math/iq_kernels.cu)
target_include_directories(cuda_iq_math PRIVATE include "${cuda_ggml_source_SOURCE_DIR}/ggml/src")
target_compile_definitions(cuda_iq_math PRIVATE GUILD_CUDA_MOE_MATH_NS=guild::cuda_moe_math)
target_link_libraries(cuda_iq_math PUBLIC CUDA::cudart)
set_target_properties(cuda_iq_math PROPERTIES POSITION_INDEPENDENT_CODE ON CUDA_STANDARD 17 CUDA_RUNTIME_LIBRARY Shared)
add_library(cuda_prefill_math STATIC)
guild_cuda_attach_prefill(cuda_prefill_math)
target_sources(cuda_prefill_math PRIVATE "${GUILD_CUDA_PREFILL_ROOT}/src/prefill/moe_fused.cu")
target_compile_definitions(cuda_prefill_math PRIVATE
  GUILD_CUDA_MOE_MATH_NS=guild::cuda_moe_math
  GUILD_CUDA_ATTN_MATH_NS=guild::cuda_attn_math)
set(_prefill_has_pre75 FALSE)
foreach(_arch IN LISTS CMAKE_CUDA_ARCHITECTURES)
  string(REGEX REPLACE "-.*$" "" _base "${_arch}")
  if(_base MATCHES "^[0-9]+$" AND _base LESS 75)
    set(_prefill_has_pre75 TRUE)
  endif()
endforeach()
if(GUILD_EXPERIMENTAL_SM60 OR _prefill_has_pre75)
  target_compile_definitions(cuda_prefill_math PRIVATE GUILD_EXPERIMENTAL_SM60=1)
endif()
target_link_libraries(cuda_prefill_math PUBLIC cuda_moe_math cuda_attn_math Guild::KernelSDK CUDA::cublas CUDA::cudart)
set_target_properties(cuda_prefill_math PROPERTIES POSITION_INDEPENDENT_CODE ON CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON CUDA_RUNTIME_LIBRARY Shared)
add_library(guild_cuda_plugin SHARED src/router.cu)
guild_configure_gkernel_plugin(guild_cuda_plugin)
target_compile_definitions(guild_cuda_plugin PRIVATE GUILD_ENABLE_CUDA=1
  GUILD_EXT_PLUGIN_VERSION_STR="${PROJECT_VERSION}" GUILD_EXT_PLUGIN_VERSION_MINOR=${PROJECT_VERSION_MINOR})
set_source_files_properties(src/router.cu PROPERTIES COMPILE_OPTIONS "$<$<COMPILE_LANGUAGE:CUDA>:--use_fast_math>")
target_link_libraries(guild_cuda_plugin PRIVATE cuda_moe_math cuda_dense_math cuda_attn_math cuda_iq_math CUDA::cudart)
set_target_properties(cuda_dense_math cuda_attn_math cuda_moe_math cuda_iq_math guild_cuda_plugin PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON CUDA_RUNTIME_LIBRARY Shared)
if(UNIX)
  target_link_options(guild_cuda_plugin PRIVATE -Wl,-Bsymbolic -Wl,--no-undefined)
endif()
set_target_properties(guild_cuda_plugin PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
include(CTest)
if(BUILD_TESTING)
  add_executable(cuda_router_test tests/router_test.cpp)
  target_link_libraries(cuda_router_test PRIVATE guild_cuda_plugin Guild::KernelSDK CUDA::cudart)
  add_test(NAME cuda_router COMMAND cuda_router_test)
  set_tests_properties(cuda_router PROPERTIES RUN_SERIAL TRUE)
  add_executable(cuda_dense_test tests/dense_test.cpp)
  target_include_directories(cuda_dense_test PRIVATE include)
  target_link_libraries(cuda_dense_test PRIVATE guild_cuda_plugin Guild::KernelSDK CUDA::cudart)
  target_compile_features(cuda_dense_test PRIVATE cxx_std_17)
  add_test(NAME cuda_dense COMMAND cuda_dense_test)
  set_tests_properties(cuda_dense PROPERTIES RUN_SERIAL TRUE)
  add_executable(cuda_attn_test tests/attn_test.cpp)
  target_include_directories(cuda_attn_test PRIVATE include)
  target_link_libraries(cuda_attn_test PRIVATE guild_cuda_plugin Guild::KernelSDK CUDA::cudart)
  target_compile_features(cuda_attn_test PRIVATE cxx_std_17)
  add_test(NAME cuda_attn COMMAND cuda_attn_test)
  set_tests_properties(cuda_attn PROPERTIES RUN_SERIAL TRUE)
  add_executable(cuda_moe_test tests/moe_test.cpp)
  target_include_directories(cuda_moe_test PRIVATE include)
  target_link_libraries(cuda_moe_test PRIVATE guild_cuda_plugin Guild::KernelSDK CUDA::cudart)
  target_compile_features(cuda_moe_test PRIVATE cxx_std_17)
  add_test(NAME cuda_moe COMMAND cuda_moe_test)
  set_tests_properties(cuda_moe PROPERTIES RUN_SERIAL TRUE)
  add_executable(cuda_prefill_test tests/prefill_test.cpp)
  target_include_directories(cuda_prefill_test PRIVATE include)
  target_link_libraries(cuda_prefill_test PRIVATE cuda_prefill_math cuda_moe_math cuda_dense_math cuda_attn_math cuda_iq_math Guild::KernelSDK CUDA::cublas CUDA::cudart)
  target_compile_features(cuda_prefill_test PRIVATE cxx_std_17)
  add_test(NAME cuda_prefill COMMAND cuda_prefill_test)
  set_tests_properties(cuda_prefill PROPERTIES RUN_SERIAL TRUE)
  add_library(cuda_c_contract OBJECT tests/cuda_contract.c)
  target_link_libraries(cuda_c_contract PRIVATE Guild::KernelSDK)
endif()
set(_sm_list "")
foreach(_arch IN LISTS CMAKE_CUDA_ARCHITECTURES)
  string(REGEX REPLACE "-.*$" "" _sm "${_arch}")
  if(NOT _sm MATCHES "^[0-9]+$")
    message(FATAL_ERROR "Package metadata requires explicit numeric CUDA architectures")
  endif()
  list(APPEND _sm_list "${_sm}")
endforeach()
list(SORT _sm_list COMPARE NATURAL)
list(GET _sm_list 0 PACKAGE_MIN_SM)
list(GET _sm_list -1 PACKAGE_MAX_SM)
configure_file(package.json.in ${CMAKE_CURRENT_BINARY_DIR}/package.json @ONLY)
install(TARGETS guild_cuda_plugin LIBRARY DESTINATION lib RUNTIME DESTINATION lib)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/package.json DESTINATION . RENAME .gkmetadata)
install(FILES LICENSE DESTINATION licenses RENAME guild-kernels-cuda-MIT.txt)
install(FILES "${cuda_ggml_source_SOURCE_DIR}/LICENSE" DESTINATION licenses RENAME GGML-MIT.txt)
