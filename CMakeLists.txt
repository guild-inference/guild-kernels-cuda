cmake_minimum_required(VERSION 3.24)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 61)
endif()
project(guild_kernels_cuda VERSION 1.0.0 LANGUAGES C CXX CUDA)
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
  OUTPUT_VARIABLE PACKAGE_SOURCE_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
configure_file(package.json.in ${CMAKE_CURRENT_BINARY_DIR}/package.json @ONLY)
find_package(GuildKernelSDK 1.1 REQUIRED CONFIG)
find_package(CUDAToolkit REQUIRED)
add_library(guild_cuda_plugin SHARED src/router.cu)
guild_configure_gkernel_plugin(guild_cuda_plugin)
target_compile_definitions(guild_cuda_plugin PRIVATE GUILD_ENABLE_CUDA=1
  GUILD_EXT_PLUGIN_VERSION_STR="${PROJECT_VERSION}")
target_compile_options(guild_cuda_plugin PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--use_fast_math>)
target_link_libraries(guild_cuda_plugin PRIVATE CUDA::cudart)
set_target_properties(guild_cuda_plugin PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
include(CTest)
if(BUILD_TESTING)
  add_executable(cuda_router_test tests/router_test.cpp)
  target_link_libraries(cuda_router_test PRIVATE guild_cuda_plugin Guild::KernelSDK CUDA::cudart)
  add_test(NAME cuda_router COMMAND cuda_router_test)
  set_tests_properties(cuda_router PROPERTIES RUN_SERIAL TRUE)
endif()
install(TARGETS guild_cuda_plugin LIBRARY DESTINATION lib RUNTIME DESTINATION lib)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/package.json DESTINATION . RENAME .gkmetadata)
